/** @fileoverview Compares source identity and Turbo build inputs in an isolated current-source copy. */
import { execFile } from 'node:child_process'
import { copyFile, mkdir, mkdtemp, readFile, rm, writeFile } from 'node:fs/promises'
import { dirname, join, resolve } from 'node:path'
import { promisify } from 'node:util'
import { sourceIdentity } from '../../clients/desktop/app/tools/build/identity'

const root = resolve('.')
const directory = await mkdtemp(join(root, '.local/final-state-audit/input-probe-'))
const execute = promisify(execFile)
const invocation = {
  cwd: directory,
  encoding: 'utf8' as const,
  windowsHide: true,
  maxBuffer: 16 * 1024 * 1024,
}
try {
  const inventory: { current: string[] } = JSON.parse(
    await readFile(join(root, '.local/final-state-audit/inventory.json'), 'utf8'),
  )
  for (const file of inventory.current) {
    await mkdir(dirname(join(directory, file)), { recursive: true })
    await copyFile(join(root, file), join(directory, file))
  }
  await execute('git', ['init', '--initial-branch=main'], invocation)
  await execute('git', ['add', '.'], invocation)

  async function turboIdentity() {
    const { stdout } = await execute(
      process.execPath,
      [join(root, 'node_modules/turbo/bin/turbo'), 'run', 'build', '--filter=@kisaki3/desktop', '--dry=json'],
      invocation,
    )
    const plan = JSON.parse(stdout)
    return plan.tasks.find((task: { taskId: string }) => task.taskId === '@kisaki3/desktop#build').hash
  }
  const baseline = { source: await sourceIdentity(directory), turbo: await turboIdentity() }
  const checks = []
  for (const file of [
    'clients/desktop/app/assets/icons/README.md',
    'clients/desktop/app/tools/dev/cli.ts',
    'clients/desktop/app/electron-builder.yml',
    'packages/tooling/README.md',
    'packages/tooling/config/typescript/tsconfig.tests.json',
    'packages/eslint-config/src/base.ts',
    '.node-version',
    '.npmrc',
    'clients/desktop/app/src/main/entry.ts',
  ]) {
    const path = join(directory, file)
    const original = await readFile(path)
    try {
      await writeFile(path, Buffer.concat([original, Buffer.from('\n')]))
      const current = { source: await sourceIdentity(directory), turbo: await turboIdentity() }
      checks.push({ file, sourceChanged: current.source !== baseline.source, turboChanged: current.turbo !== baseline.turbo })
    } finally {
      await writeFile(path, original)
    }
  }
  await writeFile(join(root, '.local/final-state-audit/input-identity-probe.json'), JSON.stringify({ node: process.version, baseline, checks }, null, 2))
  console.info(JSON.stringify(checks, null, 2))
} finally {
  await rm(directory, { recursive: true, force: true })
}
